// Drizzle schema — the single source of truth for the data model. SERVER-ONLY.
//
// Product model materialized for step 5.4 Contract definition from docs/ONTOLOGY.md.
// Better Auth's `user` is also the canonical product User. `session` is auth;
// `sessions` is the durable coaching Session. The example table is a temporary implementation reference.
// Custom database triggers and evidence grants: drizzle/0006_product_integrity.sql.
// NO FORCE RLS: architecture Deletion and Cascade requires owner-run cascades.

// Tenant scoping is two-layer per `data-layer.md`:
//   - Layer 1: every handler enters via `withTenantDb`, which binds
//     `set_config('app.user_id', …, true)` transaction-local.
//   - Layer 2: every tenant-owned table declares a `pgPolicy` reading
//     `nullif(current_setting('app.user_id', true), '')::uuid` and `.enableRLS()`.
//
// The runtime role is `coach_app_rt` (`NOBYPASSRLS`), a member of the
// `authenticated` role; `DATABASE_URL_OWNER` is migrations-only. See
// `drizzle/0001_coach_app_rt_grants.sql` for the runtime-role contract.

const TENANT_USING = sql`user_id = nullif(current_setting('app.user_id', true), '')::uuid`

// ---------------------------------------------------------------------------
// Better Auth: `user` — also the canonical ontology User profile and billing row
// ---------------------------------------------------------------------------

export const user = pgTable(
  'user',
  {
    id: uuid('id').primaryKey().defaultRandom(),
    name: text('name'),
    email: text('email').notNull().unique(),
    emailVerified: boolean('email_verified').notNull().default(false),
    image: text('image'),

    firstName: text('first_name'),
    lastName: text('last_name'),
    aboutMe: text('about_me').notNull().default(''),
    timezone: text('timezone').notNull().default('UTC'),
    subscriptionStatus: subscriptionStatus('subscription_status').notNull().default('incomplete'),
    subscriptionCancelAtPeriodEnd: boolean('subscription_cancel_at_period_end')
      .notNull()
      .default(false),
    subscriptionCurrentPeriodEnd: timestamp('subscription_current_period_end', {
      withTimezone: true,
    }),
    stripeCustomerId: text('stripe_customer_id').unique(),

    createdAt: timestamp('created_at', { withTimezone: true }).notNull().defaultNow(),
    updatedAt: timestamp('updated_at', { withTimezone: true }).notNull().defaultNow(),
  },
  (table) => [
    check(
      'user_subscription_status_check',
      sql`${table.subscriptionStatus} in ('incomplete', 'active', 'past_due', 'canceled')`,
    ),
    pgPolicy('user_self_isolation', {
      for: 'all',
      to: 'authenticated',
      using: sql`${table.id} = nullif(current_setting('app.user_id', true), '')::uuid`,
      withCheck: sql`${table.id} = nullif(current_setting('app.user_id', true), '')::uuid`,
    }),
  ],
).enableRLS()

// ---------------------------------------------------------------------------
// Better Auth `oauthProvider` plugin: `oauth_client` (global — no RLS)
// ---------------------------------------------------------------------------
//
// Columns mirror the `@better-auth/oauth-provider@1.6.22` `oauthClient` model
// (its DCR insert writes every one of these — an absent column 500s
// registration). Verified against the plugin's exported `schema` object AND
// the canonical OAuth-provider docs. This is a GLOBAL registry, not
// tenant-owned: clients are looked up by `client_id` across users during
// authorize/token, and unauthenticated DCR clients (the MCP case) have a NULL
// `user_id`. So no RLS here — `user_id` is optional ownership metadata, not a
// tenant-scoping key. Array-typed plugin fields (`scopes`, `redirect_uris`,
// `grant_types`, …) are native Postgres `text[]` columns (`.array()`): the
// Drizzle adapter runs with `supportsArrays: true` for `pg`, so it passes real
// JS arrays straight through — a scalar `text` column would round-trip a
// stringified array and break the plugin's `scopes.join(' ')`.

export const oauthClient = pgTable(
  'oauth_client',
  {
    id: uuid('id').primaryKey().defaultRandom(),
    clientId: text('client_id').notNull().unique(),
    clientSecret: text('client_secret'),
    // Optional in the plugin model — a DCR client may omit them, so nullable
    // (a NOT NULL here would 500 registration).
    name: text('name'),
    type: text('type'),
    disabled: boolean('disabled').notNull().default(false),
    // Ownership: NULL for unauthenticated DCR clients (the MCP case).
    userId: uuid('user_id'),
    referenceId: text('reference_id'),
    // Consent / session behaviour.
    skipConsent: boolean('skip_consent'),
    enableEndSession: boolean('enable_end_session'),
    subjectType: text('subject_type'),
    // OAuth registration metadata. `string[]` fields → native `text[]` (see header).
    scopes: text('scopes').array(),
    redirectUris: text('redirect_uris').array().notNull(),
    postLogoutRedirectUris: text('post_logout_redirect_uris').array(),
    grantTypes: text('grant_types').array(),
    responseTypes: text('response_types').array(),
    tokenEndpointAuthMethod: text('token_endpoint_auth_method'),
    public: boolean('public'),
    requirePKCE: boolean('require_pkce'),
    // RFC 7591 client display / legal metadata.
    uri: text('uri'),
    icon: text('icon'),
    contacts: text('contacts').array(),
    tos: text('tos'),
    policy: text('policy'),
    softwareId: text('software_id'),
    softwareVersion: text('software_version'),
    softwareStatement: text('software_statement'),
    metadata: jsonb('metadata'),
    createdAt: timestamp('created_at', { withTimezone: true }).notNull().defaultNow(),
    updatedAt: timestamp('updated_at', { withTimezone: true }).notNull().defaultNow(),
  },
  (table) => [
    index('oauth_client_client_id_idx').on(table.clientId),
    index('oauth_client_user_id_idx').on(table.userId),
  ],
)

// Columns mirror the plugin's `oauthAccessToken` model. `token` is nullable
// (the model marks it unique but not required — with the `jwt()` plugin the

    userId: uuid('user_id').references(() => user.id, { onDelete: 'cascade' }),
    referenceId: text('reference_id'),
    scopes: text('scopes').array().notNull(),
    expiresAt: timestamp('expires_at', { withTimezone: true }).notNull(),
    createdAt: timestamp('created_at', { withTimezone: true }).notNull().defaultNow(),
  },
  (table) => [
    index('oauth_access_token_user_id_idx').on(table.userId),
    index('oauth_access_token_client_id_idx').on(table.clientId),
    index('oauth_access_token_session_id_idx').on(table.sessionId),
    index('oauth_access_token_refresh_id_idx').on(table.refreshId),
    pgPolicy('oauth_access_token_tenant_isolation', {
      for: 'all',
      to: 'authenticated',
      using: TENANT_USING,
      withCheck: TENANT_USING,
    }),
  ],
).enableRLS()

// ---------------------------------------------------------------------------
// Product tenant keys are server-supplied. Composite foreign keys prevent a
// child from claiming a different tenant from its parent even under owner SQL.
// Evidence session links use NO ACTION; whole-account deletion removes both
// sides together. Action links instead clear session_id (F5 Actions).
const tenantPolicy = (name: string) =>
  pgPolicy(`${name}_tenant_isolation`, {
    for: 'all',
    to: 'authenticated',
    using: TENANT_USING,
    withCheck: TENANT_USING,
  })
const productId = () => uuid('id').primaryKey().defaultRandom()
const tenantId = () =>
  uuid('user_id')
    .notNull()
    .references(() => user.id, { onDelete: 'cascade' })
const created = () => timestamp('created_at', { withTimezone: true }).notNull().defaultNow()
const updated = () => timestamp('updated_at', { withTimezone: true }).notNull().defaultNow()

export const cycles = pgTable(
  'cycles',
  {
    id: productId(),
    userId: tenantId(),
    intention: text('intention'),
    startAt: date('start_at').notNull(),
    endAt: date('end_at').notNull(),
    status: cycleStatus('status').notNull().default('active'),
    createdAt: created(),
    updatedAt: updated(),
  },
  (t) => [
    uniqueIndex('cycles_id_user_uidx').on(t.id, t.userId),
    index('cycles_user_id_idx').on(t.userId),
    uniqueIndex('cycles_one_active_uidx')
      .on(t.userId)
      .where(sql`${t.status} = 'active'`),
    check('cycles_date_range_check', sql`${t.endAt} > ${t.startAt}`),
    tenantPolicy('cycles'),
  ],
).enableRLS()

export const results = pgTable(
  'results',
  {
    id: productId(),
    userId: tenantId(),
    objectiveId: uuid('objective_id').notNull(),
    title: text('title').notNull(),
    description: text('description'),
    metricType: metricType('metric_type').notNull(),
    startValue: numeric('start_value', { mode: 'number' }),
    targetValue: numeric('target_value', { mode: 'number' }),
    unit: text('unit'),
    cadence: habitCadence('cadence'),
    per: integer('per'),
    since: date('since'),
    status: resultStatus('status').notNull().default('in_progress'),
    createdAt: created(),
    updatedAt: updated(),
  },
  (t) => [
    uniqueIndex('results_id_user_uidx').on(t.id, t.userId),
    index('results_user_id_idx').on(t.userId),
    index('results_objective_user_idx').on(t.objectiveId, t.userId),
    foreignKey({
      name: 'results_objective_tenant_fk',
      columns: [t.objectiveId, t.userId],
      foreignColumns: [objectives.id, objectives.userId],
    }).onDelete('cascade'),
    check(
      'results_number_check',
      sql`${t.metricType} <> 'number' OR (${t.startValue} IS NOT NULL AND ${t.targetValue} IS NOT NULL AND ${t.cadence} IS NULL AND ${t.per} IS NULL AND ${t.since} IS NULL)`,
    ),
    check(
      'results_percentage_check',
      sql`${t.metricType} <> 'percentage' OR (${t.startValue} IS NOT NULL AND ${t.targetValue} IS NOT NULL AND ${t.startValue} BETWEEN 0 AND 100 AND ${t.targetValue} BETWEEN 0 AND 100 AND ${t.cadence} IS NULL AND ${t.per} IS NULL AND ${t.since} IS NULL AND ${t.unit} IS NULL)`,
    ),
    check(
      'results_milestone_check',
      sql`${t.metricType} <> 'milestone' OR (${t.startValue} IS NOT NULL AND ${t.startValue} = 0 AND ${t.targetValue} IS NOT NULL AND ${t.targetValue} = 1 AND ${t.cadence} IS NULL AND ${t.per} IS NULL AND ${t.since} IS NULL AND ${t.unit} IS NULL)`,
    ),
    check(
      'results_habit_check',
      sql`${t.metricType} <> 'habit' OR (${t.cadence} IS NOT NULL AND ${t.per} IS NOT NULL AND ${t.per} >= 1 AND ${t.since} IS NOT NULL AND ${t.startValue} IS NULL AND ${t.targetValue} IS NULL AND ${t.unit} IS NULL)`,
    ),
    check(
      'results_performance_check',
      sql`${t.metricType} <> 'performance' OR (${t.targetValue} IS NOT NULL AND ${t.since} IS NOT NULL AND ${t.cadence} IS NULL AND ${t.per} IS NULL)`,
    ),
    check(
      'results_distinct_target_check',
      sql`${t.metricType} NOT IN ('number', 'percentage') OR ${t.targetValue} <> ${t.startValue}`,
    ),
    check(
      'results_finite_values_check',
      sql`(${t.startValue} IS NULL OR ${t.startValue} NOT IN ('NaN'::numeric, 'Infinity'::numeric, '-Infinity'::numeric)) AND (${t.targetValue} IS NULL OR ${t.targetValue} NOT IN ('NaN'::numeric, 'Infinity'::numeric, '-Infinity'::numeric))`,
    ),
    tenantPolicy('results'),
  ],
).enableRLS()

    createdAt: created(),
    updatedAt: updated(),
  },
  (t) => [
    uniqueIndex('actions_id_user_uidx').on(t.id, t.userId),
    index('actions_user_id_idx').on(t.userId),
    index('actions_cycle_user_idx').on(t.cycleId, t.userId),
    index('actions_objective_cycle_user_idx').on(t.objectiveId, t.cycleId, t.userId),
    index('actions_session_user_idx').on(t.sessionId, t.userId),
    index('actions_outstanding_order_idx').on(
      t.cycleId,
      t.objectiveId,
      t.sortOrder.asc().nullsLast(),

    ),
